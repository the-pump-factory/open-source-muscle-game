vars.cam.rotation = LookRotation((vars.player.position + vec(0, 1.5, 0)) - vars.cam.position)

turnSpeed = 250 * vars.deltaTime
moveSpeed = 1.3
animSpeedMul = 1.5
animMixClamp = 1.05
lookChance = 0.0002
poseSpeed = 4

moveX = InputManager.moveAxisX
moveY = InputManager.moveAxisY

camRot = vars.cam.rotation
forward = vec(0, 0, 1)
camDir = camRot * forward
camDir = vec(camDir.x, 0, camDir.z)
camDir = camDir / camDir.magnitude
inputAlign = EulerQuaternion(0, SignedAngle(forward, camDir, vec(0, 1, 0)), 0)

input = vec(moveX, 0, moveY)
inputMag = input.magnitude

isPressed = inputMag > 0

if isPressed then

playerDir = vars.player.rotation * forward
playerDir = vec(playerDir.x, 0, playerDir.z)
playerDir = playerDir / playerDir.magnitude

inputDir = input / inputMag
targetDir = inputAlign * inputDir

inputDir = inputAlign * inputDir

rotAngPlayer = SignedAngle(playerDir, targetDir, vec(0, 1, 0))
rotAngPlayer = clamp(rotAngPlayer, -turnSpeed, turnSpeed) 
vars.player.rotation = EulerQuaternion(0, rotAngPlayer, 0) * vars.player.rotation

inputMag = inputMag * 1.4
if inputMag > 1 then
inputMag = 1
end if
moveSpeed = moveSpeed * inputMag
vars.walkLayer.mix = min(animMixClamp, moveSpeed)
vars.walkController.speed = moveSpeed * animSpeedMul
vars.player.position = vars.player.position + (inputDir * moveSpeed * vars.deltaTime)
else
vars.walkLayer.mix = 0
end if

if InputManager.jumpButton then
vars.posing = clamp(vars.posing + vars.deltaTime * poseSpeed, 0, 1)
else
vars.posing = clamp(vars.posing - vars.deltaTime * poseSpeed, 0, 1)
end if
vars.idleLayer.mix = max(0, (1 - vars.walkLayer.mix) * (1 - vars.posing))
vars.poseLayer.mix = max(0, (1 - vars.walkLayer.mix) * vars.posing)


if vars.isLooking then
if vars.lookState.time < vars.lookTime then
// stop looking
vars.lookLayer.mix = 0
vars.lookController.speed = 0
vars.lookState.time = 0
vars.lookTime = 0
vars.isLooking = false
else
vars.lookTime = vars.lookState.time
end if
else if vars.lookCooldown > 0 then
vars.lookCooldown = vars.lookCooldown - vars.deltaTime
else if RNG.Range(0, 1) < lookChance then
vars.lookLayer.mix = 1
vars.lookController.speed = 1
vars.lookState.time = 0
vars.lookTime = 0
vars.isLooking = true
vars.lookCooldown = vars.lookCooldownStart
end if
